
cd ~/research/chance-2048 && git pull && git status -sb 
tmux new -s ppo


jobs=(); for seed in 0 1 2 3 4; do for level in S0 S1 S2 S3 S4; do jobs+=("$level $seed"); done; done
for k in $(seq 0 12); do
  ( core=$((40+k))
    for ((j=k; j<25; j+=13)); do
      set -- ${jobs[j]}; level=$1 seed=$2
      scripts/launch.sh --image chance-2048-ppo:0.1 --cpus $core,$((core+60)) \
        --name E003-ppo-$level-s$seed -- \
        python -m chance2048.train_ppo --config configs/ppo.yaml \
          --level $level --seed $seed --exp E003 \
      && scripts/launch.sh --image chance-2048-ppo:0.1 --cpus $core,$((core+60)) \
        --name E003-ppo-$level-s$seed-eval -- \
        python -m chance2048.evaluate_ppo --run $(ls -d runs/E003-ppo-$level-s$seed-* | tail -1)
    done ) > ~/research/chance-2048/runs/_logs/E003-ppo-queue-$k.log 2>&1 &
done

docker ps --filter name=E003-ppo | wc -l

docker run -d --rm --name tb --user "$(id -u):$(id -g)" \
  -p 127.0.0.1:6006:6006 -v "$PWD/runs:/runs:ro" \
  chance-2048-lz:0.2 tensorboard --logdir /runs --host 0.0.0.0 --port 6006

grep -h "rollout/ep_rew_mean\|ep_score" runs/E003-ppo-S0-s0-*/metrics.csv | tail -3
