launch() {
  scripts/launch.sh --image chance-2048-lz:0.2 --gpus "$5" \
    --cpus "$6,$(($6+1)),$(($6+60)),$(($6+61))" ${7:+--deadline "$7"} \
    --name "E003-$1-$3-s$4" --detach \
    -- python -m chance2048.train_lightzero --config "configs/$2.yaml" --level "$3" --seed "$4" --exp E003
}

cpus=(34 36 38 40 42 44 46 48); i=0
for seed in 2 3; do for level in S0 S1 S2 S4; do
  launch mz mz $level $seed $seed ${cpus[$i]}; i=$((i+1))
done; done
