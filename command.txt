evalrun() {
  local run=$(ls -d runs/$1-* | grep -v eval | tail -1)
  scripts/launch.sh --image chance-2048-lz:0.2 --gpus "$2" \
    --cpus "$3,$(($3+1)),$(($3+60)),$(($3+61))" --name "E003-eval-${1#E003-}" --detach \
    -- python -m chance2048.evaluate --run "$run"
}

python3 -I - runs <<'EOF'
import json, sys
from pathlib import Path
train, evals = {}, {}
for f in Path(sys.argv[1]).glob("E003-*/run.json"):
    r = json.loads(f.read_text())
    if r["tags"].get("kind") == "evaluation":
        evals.setdefault(r["tags"]["source_run"], []).append(r["status"])
    elif r["status"] == "completed":
        train[(r["name"], r["seed"])] = r["run_id"]
missing = 0
for (name, seed), rid in sorted(train.items()):
    st = evals.get(rid, [])
    ok = "completed" in st
    missing += not ok
    print(f"{'ok ' if ok else 'MISSING'}  {name:<20} s{seed}  eval: {', '.join(st) or 'none'}")
print(f"\n{len(train)} completed training runs, {missing} without a completed evaluation")
EOF
