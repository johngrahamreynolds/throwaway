evalrun() {
  local run=$(ls -d runs/$1-* | grep -v eval | tail -1)
  scripts/launch.sh --image chance-2048-lz:0.2 --gpus "$2" \
    --cpus "$3,$(($3+1)),$(($3+60)),$(($3+61))" --name "E003-eval-${1#E003-}" --detach \
    -- python -m chance2048.evaluate --run "$run"
}
