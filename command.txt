
cd ~/research/chance-2048
export MACHINE_NAME=ws-02

grep -H '"status"' runs/E003-mz-S*-s[23]-*/run.json

cpus=(34 36 38 40 42 44 46 48); i=0
for seed in 2 3; do for level in S0 S1 S2 S4; do
  run=$(ls -d runs/E003-mz-$level-s$seed-* | tail -1)
  c=${cpus[$i]}; i=$((i+1))
  scripts/launch.sh --image chance-2048-lz:0.2 --gpus $seed \
    --cpus "$c,$((c+1)),$((c+60)),$((c+61))" --name "E003-eval-mz-$level-s$seed" --detach \
    -- python -m chance2048.evaluate --run "$run"
done; done

launch() {
  scripts/launch.sh --image chance-2048-lz:0.2 --gpus "$5" \
    --cpus "$6,$(($6+1)),$(($6+60)),$(($6+61))" ${7:+--deadline "$7"} \
    --name "E003-$1-$3-s$4" --detach \
    -- python -m chance2048.train_lightzero --config "configs/$2.yaml" --level "$3" --seed "$4" --exp E003
}

launch mz mz S2 4 2 50; launch mz mz S4 4 3 52

docker ps --filter name=E003 --format '{{.Names}}\t{{.Status}}'
